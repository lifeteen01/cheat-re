cmake_minimum_required(VERSION 3.20)
project(re_freestanding_gate LANGUAGES CXX ASM)
# Also supports the locally extracted C++-only GNU toolchain (no cc1).
set(CMAKE_ASM_COMPILE_OBJECT
    "<CMAKE_CXX_COMPILER> <DEFINES> <INCLUDES> <FLAGS> -x c++ -E -P <SOURCE> -o <OBJECT>.s"
    "<CMAKE_ASM_COMPILER> <DEFINES> <INCLUDES> <FLAGS> -c <OBJECT>.s -o <OBJECT>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Recovered gate requires x64")
endif()
if(WIN32)
    add_library(re_gate SHARED gate.cpp gate_x64.S)
    set_target_properties(re_gate PROPERTIES PREFIX "")
    target_link_options(re_gate PRIVATE -nostdlib -nodefaultlibs -nostartfiles
        -Wl,--entry,ReGateDllEntry,--no-undefined,--exclude-all-symbols)
else()
    add_library(re_gate STATIC gate.cpp gate_x64.S)
endif()
target_compile_features(re_gate PRIVATE cxx_std_20)
target_compile_definitions(re_gate PRIVATE RE_GATE_BUILD)
target_compile_options(re_gate PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror;-O2;-ffreestanding;-fno-builtin;-fno-exceptions;-fno-rtti;-fno-stack-protector;-fno-threadsafe-statics;-fno-use-cxa-atexit;-fno-unwind-tables;-fno-asynchronous-unwind-tables;-fno-tree-loop-distribute-patterns>")
add_executable(owned_gate_checks checks.cpp checks_x64.S)
target_compile_features(owned_gate_checks PRIVATE cxx_std_20)
target_compile_options(owned_gate_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
if(WIN32)
    target_link_options(owned_gate_checks PRIVATE -static -static-libgcc -static-libstdc++)
    add_test(NAME owned_gate COMMAND owned_gate_checks "$<TARGET_FILE:re_gate>")
else()
    target_link_libraries(owned_gate_checks PRIVATE re_gate)
    add_test(NAME owned_gate COMMAND owned_gate_checks)
endif()
