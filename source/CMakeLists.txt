cmake_minimum_required(VERSION 3.20)
project(re_base LANGUAGES CXX)
option(RE_BUILD_WINDOWS_BASE "Build the Windows x64 platform DLL and owned host test" ${WIN32})
option(RE_BUILD_OFFLINE_RECONSTRUCTION "Build the full offline reconstruction and fixtures" OFF)
option(RE_BUILD_TEST_SCENE "Build the owned reconstructed internal menu and ESP test scene" OFF)
option(RE_BUILD_CHEAT_DLL "Build the full reconstructed cheat DLL and hooks DLL" ${WIN32})
if(RE_BUILD_WINDOWS_BASE)
    add_subdirectory(windows_base)
endif()
if(RE_BUILD_CHEAT_DLL)
    add_subdirectory(cheat_dll)
endif()
if(RE_BUILD_OFFLINE_RECONSTRUCTION)
    add_subdirectory(reconstructed)
endif()
enable_testing()
if(RE_BUILD_TEST_SCENE)
    add_subdirectory(test_scene)
endif()
